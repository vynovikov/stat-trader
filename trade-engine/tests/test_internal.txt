@pytest.mark.parametrize(
    "case",
    check_off_market_tp_test_cases,
    ids=[c["name"] for c in check_off_market_tp_test_cases],
)
def test_check_off_market_tp(case):
    # Initialization
    internal = Internal(trend=case["initial_trend"], payload=case["initial_payload"])

    # Check TP
    internal.check_off_market_tp(candle=case["candle"])

    # Assertions
    assert internal.payload.direction == case["expected_payload"].direction
    assert round(internal.payload.sl, 2) == round(case["expected_payload"].sl, 2)
    assert round(internal.payload.tp, 2) == round(case["expected_payload"].tp, 2)
